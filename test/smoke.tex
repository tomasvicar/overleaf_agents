\section{Algorithm}

\begin{algorithm}[H]
  \caption{Threshold sweep}
  \begin{algorithmic}[1]
    \State $best \gets -\infty$
    \For{$\theta \in \Theta$}
      \If{$f(\theta) > best$} \State $best \gets f(\theta)$ \EndIf
    \EndFor
    \State \Return $best$
  \end{algorithmic}
\end{algorithm}
